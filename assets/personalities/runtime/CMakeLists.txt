cmake_minimum_required(VERSION 3.25)
project(AudioHoleAvatar VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
option(AUDIOHOLE_AVATAR_BUILD_VIEWER "Build standalone wxWidgets viewer" ON)
option(AUDIOHOLE_AVATAR_BUILD_TESTS "Build avatar module tests" ${BUILD_TESTING})
find_package(Threads REQUIRED)
if(NOT TARGET nlohmann_json::nlohmann_json)
  find_package(nlohmann_json 3.11.3 REQUIRED)
endif()
add_library(audiohole_avatar src/assets.cpp src/sha256.cpp src/player.cpp src/session.cpp)
add_library(audiohole::avatar ALIAS audiohole_avatar)
target_compile_features(audiohole_avatar PUBLIC cxx_std_20)
target_include_directories(audiohole_avatar PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_link_libraries(audiohole_avatar PUBLIC Threads::Threads PRIVATE nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(audiohole_avatar PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(audiohole_avatar PRIVATE -Wall -Wextra -Wpedantic)
endif()

# Reuse the parent application's wx targets. Standalone builds use an installed
# development package; this module never downloads or chooses a new GUI framework.
if(NOT TARGET wx::core)
  find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base)
  include(${wxWidgets_USE_FILE})
  add_library(audiohole_avatar_wx_dependencies INTERFACE)
  target_link_libraries(audiohole_avatar_wx_dependencies INTERFACE ${wxWidgets_LIBRARIES})
  set(avatar_wx_dependencies audiohole_avatar_wx_dependencies)
else()
  set(avatar_wx_dependencies wx::core wx::base)
endif()
add_library(audiohole_avatar_wx src/native_decoder.cpp src/wx_panel.cpp)
add_library(audiohole::avatar_wx ALIAS audiohole_avatar_wx)
target_compile_features(audiohole_avatar_wx PUBLIC cxx_std_20)
target_link_libraries(audiohole_avatar_wx PUBLIC audiohole_avatar ${avatar_wx_dependencies})
if(WIN32)
  target_link_libraries(audiohole_avatar_wx PRIVATE windowscodecs shlwapi ole32)
endif()
if(AUDIOHOLE_AVATAR_BUILD_VIEWER)
  add_executable(audiohole_avatar_viewer WIN32 viewer/main.cpp)
  target_link_libraries(audiohole_avatar_viewer PRIVATE audiohole_avatar_wx nlohmann_json::nlohmann_json)
  target_include_directories(audiohole_avatar_viewer PRIVATE ../integration ../integration/inspection/src)
  target_compile_definitions(audiohole_avatar_viewer PRIVATE AVATAR_DEFAULT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/..")
endif()
if(MSVC)
  target_compile_options(audiohole_avatar_wx PRIVATE /W4 /permissive- /utf-8)
  if(TARGET audiohole_avatar_viewer)
    target_compile_options(audiohole_avatar_viewer PRIVATE /W4 /permissive- /utf-8)
    target_link_options(audiohole_avatar_viewer PRIVATE /MANIFEST:EMBED
      "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/viewer/viewer.manifest")
  endif()
elseif(WIN32 AND TARGET audiohole_avatar_viewer)
  enable_language(RC)
  target_sources(audiohole_avatar_viewer PRIVATE viewer/viewer.rc)
endif()
if(AUDIOHOLE_AVATAR_BUILD_TESTS)
  add_executable(avatar_tests tests/tests.cpp)
  target_link_libraries(avatar_tests PRIVATE audiohole_avatar_wx nlohmann_json::nlohmann_json)
  target_include_directories(avatar_tests PRIVATE ../integration ../integration/inspection/src)
  target_compile_definitions(avatar_tests PRIVATE AVATAR_TEST_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/..")
  add_test(NAME avatar_engine COMMAND avatar_tests --report ${CMAKE_CURRENT_BINARY_DIR}/validation/native-tests.json)
endif()
